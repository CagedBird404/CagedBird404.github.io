## Bellman–Ford

原文件：`算法/最短路径/单源/Bellman–Ford.cpp`

```cpp
/*
与Dijkstra相比，这个算法唯一好处在于可以计算负权边，还能检测负环

时间复杂度O(nm);
*/
#include <bits/stdc++.h> 
using namespace std;
#define int long long                 
#define endl '\n'                     
#define all(x) (x).begin(), (x).end() 

constexpr int MOD = 1e9 + 7;

struct node
{
    int x,y,z;
};
void solve()
{
    int n,m,s;
    cin >> n >> m >> s;
    vector<node> a(m);
    for(int i = 0;i < m;i++)
    {
        cin >> a[i].x >> a[i].y >> a[i].z;
    }
    vector<int> v(n+1,1e9);
    v[s] = 0;
    for(int i = 1;i <= n-1;i++)
    {
        for(auto k : a)
        {
            if(v[k.x] < 1e9 && v[k.y] > v[k.x] + k.z)
            {
                v[k.y] = v[k.x] + k.z;
            }
        }
    }

    //检测负环
    bool ok = 0;
    for(auto k : a)
    {
        if(v[k.x] < 1e9 && v[k.y] > v[k.x] + k.z)
        {
            ok = 1;
            break;
        }
    }
    if(ok) cout << "存在负环" << endl;
    else {
        for(int i = 1;i <=n;i++)
        {
            cout << v[i] << " ";
        }
    }
}

signed main()
{
    cin.tie(0)->ios::sync_with_stdio(0);                
    cout << setiosflags(ios::fixed) << setprecision(2); 
    int TT = 1;
    cin >> TT; 
    while (TT--)
    {
        solve(); 
        cout << endl;
    }
}
```

## Dijkstra

原文件：`算法/最短路径/单源/Dijkstra.cpp`

```cpp
/*
给定一个起点，找出这个起点到其他所有点的最短路径
条件：图的边权值不能为负数
每次从未确定的点中选出路径最短的点，再通过他找他的邻居

1、起点距离记为 0，其他点设为无穷大

2、每次选出一个“最近但未访问”的点

3、用它“松弛”（relax）所有邻居：

时间复杂度：O((m+n)log(n))
*/
#include <bits/stdc++.h> 
using namespace std;
#define int long long                 
#define endl '\n'                     
#define all(x) (x).begin(), (x).end() 

constexpr int MOD = 1e9 + 7;

void solve()
{
    int n, m, s;
    cin >> n >> m >> s;
    vector<vector<pair<int ,int>>> a(n + 1);
    for(int i = 0;i < m;i++)
    {
        int u, v, w;
        cin >> u >> v >> w;
        a[u].push_back({v, w});
    }

    vector<int> wf(n + 1, 1e18);
    wf[s] = 0;
    priority_queue<pair<int, int>, vector<pair<int ,int>>, greater<pair<int ,int>>> pq;
    pq.push({0, s});
    while(!pq.empty())
    {
        pair<int ,int> p = pq.top();
        pq.pop();
        int v = p.second;
        int w = p.first;
        if(w > wf[v]) continue;

        for(auto [u, f] : a[v])
        {
            if(f + w < wf[u])
            {
                wf[u] = f + w;
                pq.push({wf[u], u});
            }
        }
    }
}

signed main()
{
    cin.tie(0)->ios::sync_with_stdio(0);                
    cout << setiosflags(ios::fixed) << setprecision(2); 
    int TT = 1;
    //cin >> TT; 
    while (TT--)
    {
        solve(); 
        //cout << endl;
    }
}
```
