## 1全局函数做友元

原文件：`基础/面向对象/类和对象/4友元/1全局函数做友元.cpp`

```cpp
/*
对于一些函数和类，可以用friend修饰后
可以访问他的私有和保护属性
*/
#include<bits/stdc++.h>
using namespace std;

class Building
{

    friend void cc(Building *build); //将其设置成好朋友，使其可以访问他的私有

public:

    Building()
    {
        keting = "客厅";
        woshi = "卧室";
    }

    string keting;

private:
    string woshi;
};

void cc(Building *build)
{
    cout << build -> woshi << endl; //正常来讲，私有不可访问
    cout << build -> keting << endl;
}



int main(){
    Building bu;
    cc(&bu);
}
```

## 2类做友元

#include<bits/stdc++.h>
using namespace std;

class Building; //类的声明

class goodgay
{
    
public:

    goodgay();
    void visit();

    Building *build;
};

class Building
{
    
    friend class goodgay;

public:

    Building();
    string ke;

private: 
    
    string wo;

};
//类外写成员函数
Building::Building()
{
    ke = "keting";
    wo = "woshi";
}

goodgay::goodgay()
{
    build = new Building;//类内的指针都要开辟新空间
}

void goodgay::visit()
{
    cout << "fangwen: " << build -> ke << endl;
    cout << "fangwen: " << build -> wo << endl;
}

void text01()
{
    goodgay p1;
    p1.visit();
}

int main(){
    text01();
}
