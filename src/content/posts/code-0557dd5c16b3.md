## AC自动机

原文件：`算法/字符串/AC自动机.cpp`

```cpp
/*
先使用前缀树。再使用fail指针（用bfs）

作用是找到敏感词在文本中出现几次
*/

#include <bits/stdc++.h> 
using namespace std;
#define int long long                 
#define endl '\n'                     
#define all(x) (x).begin(), (x).end() 

constexpr int MOD = 1e9 + 7;

void solve()
{
    int n;
    cin >> n;
    vector<int> end(n);

    vector<vector<int>> tree(1, vector<int>(26));
    
    int cnt = 0;

    //建树
    for(int i = 0; i < n;i++)
    {
        string s;
        cin >> s;
        int now = 0;
        for(int j = 0; j < s.size();j++)
        {
            int c = s[j] - 'a';
            if(!tree[now][c]) 
            {
                tree.push_back(vector<int>(26));
                tree[now][c] = ++cnt;
            }
            now = tree[now][c];
        }
        end[i] = now;
    }
    //建立fail指针
    vector<int> fail(cnt + 1);
    queue<int> q;
    vector<int> orz;
    for(int i = 0;i < 26;i++)
    {
        if(tree[0][i]) q.push(tree[0][i]);
    }
    while(!q.empty())
    {
        int now = q.front();
        q.pop();
        orz.push_back(now);
        for(int i = 0;i < 26;i++)
        {
            if(!tree[now][i])
            {
                tree[now][i] = tree[fail[now]][i];
            }
            else{
                fail[tree[now][i]] = tree[fail[now]][i];
                q.push(tree[now][i]);
            }
        }
    }
    //记录词频
    string s;
    cin >> s;
    int now = 0;vector<int> times(cnt + 1);
    for(int i = 0;i < s.size();i++)
    {
        times[tree[now][s[i] - 'a']]++;
        now = tree[now][s[i] - 'a'];
    }
    //反向建图
    //累加子点的times次数
    for(int i = orz.size() - 1;i >= 0;i--)
    {
        int u = orz[i];
        times[fail[u]] += times[u];
    }
    //统计答案
    int ans = 0;
    for(int i = 0;i < n;i++)
    {
        ans += times[end[i]];
    }
    cout << ans;
}

signed main()
{
    cin.tie(0)->ios::sync_with_stdio(0);                
    cout << setiosflags(ios::fixed) << setprecision(2); 
    int TT = 1;
    //cin >> TT; 
    while (TT--)
    {
        solve(); 
        cout << endl;
    }
}
```

## KMp算法字符串

原文件：`算法/字符串/KMp算法字符串.cpp`

```cpp
/*
KMP解决的是s1字符串是否包含s2，如果包含返回s1包含s2的最左开头位置，不包含返回-1
暴力的话肯定是O(n*m) KMP是O(m + n)

next数组的含义：
    对于s2求，不包含当前位置，他的前缀和后缀最大匹配长度（不能包含整体）
        a a b a a b s a a b a a a 无
next:  -1 0 1 0 1 2 3 0 1 2 3 4 5 2 

简单来说，不匹配就跳回去，next数组就是将暴力的查询的每次建立了联系
建立next数组也一样，前两个能确定是 -1 和 0 后面就是看当前和前面前缀延伸的下一个字符是否匹配
如果是就加上1，不是就跳回再前面的前缀延伸，继续查看是否匹配

通过迭代+回溯的方式，高效算出next数组的每个值，核心是利用已有的前缀信息快速定位最长匹配，
避免重复比较。
复杂度O n+m
*/
// KMP算法优化模板
// 1. 该模板实现了KMP算法，计算了模式串的部分匹配表（next数组）。
// 2. 使用了std::vector来存储next数组，避免了数组越界的风险。
// 3. `solve()` 函数可以根据需要填充具体的字符串匹配逻辑。
#include <bits/stdc++.h> 
using namespace std;
#define int long long                 
#define endl '\n'                     
#define all(x) (x).begin(), (x).end() 

constexpr int MOD = 1e9 + 7;

void solve()
{
    string s1,s2;
    cin >> s1 >> s2;
    int n = s1.size(),m = s2.size();
    vector<int> next(m);
    int j = 0;
    for(int i = 1;i < m;i++)
    {
        while(j && s2[i] != s2[j])
        j = next[j-1];
        if(s2[i] == s2[j]) j++;
        next[i] = j;
    }
/*
对于KMP还有一种用法，每一位的p = i - next[i],如果i % p == 0 那么这就是最小的长度对于的s，若干个s可以拼接成
前面的字符串，而p就是最小单元长度
*/
    j = 0;
    for(int i = 0;i < n;i++)
    {
        while(j && (j == m || s1[i] != s2[j]))
        j = next[j-1];
        if(s1[i] == s2[j]) j++;
        if(j == m) 
        cout << i - m + 1  << endl;
    }
    for(int i = 0;i < m;i++)
    {
        cout << next[i] << " ";
    }
}

signed main()
{
    cin.tie(0)->ios::sync_with_stdio(0);                
    cout << setiosflags(ios::fixed) << setprecision(2); 
    int TT = 1;
    //cin >> TT; 
    while (TT--)
    {
        solve(); 
        //cout << endl;
    }
}
```

## sunday

原文件：`算法/字符串/sunday.cpp`

```cpp
#include<iostream>
#include<vector>
#include<string>

using namespace std;

int main()
{
    string s1, s2;
    cin >> s1 >> s2;

    int n = s1.size(), m = s2.size();
    vector<int> next(26, -1);
    // 记录模式串中字符最后出现的位置
    for (int i = 0; i < m; i++)
    {
        next[s2[i] - 'a'] = i;
    }
    int now = 0;
    while (now <= n - m)
    {
        int ok = 1;
        // 当前窗口匹配
        for (int i = 0; i < m; i++)
        {
            if (s1[now + i] != s2[i])
            {
                ok = 0;
                break;
            }
        }

        if (ok)
        {
            cout << now << endl;
        }
        if (now + m >= n)
            break;
        char c = s1[now + m];
        if (next[c - 'a'] == -1)
        {
            now += m + 1;
        }
        else
        {
            now += m - next[c - 'a'];
        }
    }

    return 0;
}
```

```cpp
constexpr int MOD = 1e9 + 7;


void solve()
{
    int n,q;
    cin >> n >> q;
    vector<int> a(n+1);
    map<int,ull> mp;
    random_device rd;
    mt19937 gen(rd());
    uniform_int_distribution<> dist(0, 9999);
    for(int i=1;i <= n;i++)
    {
        int x;cin >> x;
        a[i] = x;
        if(!mp.count(a[i])) mp[a[i]] = ((ull)dist(gen) << 32) ^ (ull)dist(gen);
    }
    vector<ull> b(n+1);
    for(int i = 1;i <= n;i++)
    {
        b[i] = b[i-1] ^ (mp[a[i]]);
    }
    while(q--)
    {
        int l,r;
        cin >> l >> r;
        if((r - l + 1) % 2 == 1)
        {
            cout << "NO" << endl;
            continue;
        }
        int s = 1;
        if((b[r] ^ b[l-1]) != 0) s = 0;
        if(s) cout << "YES" << endl;
```

## 哈希前缀字符串双

原文件：`算法/字符串/哈希前缀字符串双.cpp`

```cpp
/*
这里用于解决两个字符串的公共最长子字符串
复杂度O(n+m)log min(n,m)
*/


#include <bits/stdc++.h> 
using namespace std;
#define int long long                 
#define endl '\n'                     
#define all(x) (x).begin(), (x).end() 

typedef unsigned long long ull;

constexpr int MOD = 1e9 + 7;
const ull B1 = 131;
const ull B2 = 137;

string s1,s2;
vector<ull> h1,h2,p1,p2;
vector<ull> H1,H2,P1,P2;

pair<ull,ull> get1(int l,int r)
{
    return {h1[r] - h1[l-1] * p1[r-l+1],h2[r] - h2[l-1] * p2[r-l+1]};
}

pair<ull,ull> get2(int l,int r)
{
    return {H1[r] - H1[l-1] * P1[r-l+1],H2[r] - H2[l-1] * P2[r-l+1]};
}

bool check(int sz)
{
    if(sz == 0) return 1;
    unordered_set<ull> st;
    for(int i = 1; i <= s1.size() - sz + 1;i++)
    {
        auto [x,y] = get1(i,i + sz - 1);
        ull v = (x << 32) ^ y;
        st.insert(v);
    }
    for(int i = 1; i <= s2.size() - sz + 1;i++)
    {
        auto [x,y] = get2(i,i + sz - 1);
        ull v = (x << 32) ^ y;
        if(st.count(v)) return 1;
    }
    return 0;
}
void solve()
{
    
    cin >> s1 >> s2;
    int n = s1.size();
    int m = s2.size();
    h1.assign(n+1,0); h2.assign(n+1,0);
    p1.assign(n+1,1); p2.assign(n+1,1);
    H1.assign(m+1,0); H2.assign(m+1,0);
    P1.assign(m+1,1); P2.assign(m+1,1);
    for(int i = 1;i <= n;i++)
    {
        h1[i] = h1[i-1] * B1 + s1[i-1];
        h2[i] = h2[i-1] * B2 + s1[i-1];
        p1[i] = p1[i-1] * B1;
        p2[i] = p2[i-1] * B2;
    }
    for(int i = 1;i <= m;i++)
    {
        H1[i] = H1[i-1] * B1 + s2[i-1];
        H2[i] = H2[i-1] * B2 + s2[i-1];
        P1[i] = P1[i-1] * B1;
        P2[i] = P2[i-1] * B2;
    }
   int l = 0;
   int r = min(n,m);
   int ans = 0;
   while(l <= r)
   {
        int mid = l + (r - l)/2;
        if(check(mid))
        {
            l = mid + 1;
            ans = mid;
        } 
        else r = mid - 1;
   }
   cout << ans;
}

signed main()
{
    cin.tie(0)->ios::sync_with_stdio(0);                
    cout << setiosflags(ios::fixed) << setprecision(2); 
    int TT = 1;
    //cin >> TT; 
    while (TT--)
    {
        solve(); 
        //cout << endl;
    }
}
```

## 最小表示法

```cpp
/*
若长度为 n 的字符串 s 中可以选择一个位置 i，使得  
si ⋯sn s1 ⋯si−1 =t，则称 s 与 t 循环同构。字符串 s 的最小表示为与 s 循环同构的所有字符串中字典序最小的字符串
求最小字典序字符串
时间复杂度On
*/
#include <bits/stdc++.h> 
using namespace std;
#define int long long                 
#define endl '\n'                     
#define all(x) (x).begin(), (x).end() 

constexpr int MOD = 1e9 + 7;

void solve()
{
    string s;
    cin >> s;
    int n = s.size();
    s += s;
    int i = 0,j = 1,k = 0;
    while(i < n && j < n && k < n)
    {
        if(s[i+k] == s[j+k]) k++;
        else{
            if(s[i+k] > s[j+k]) i = i + k + 1;
            else j = j + k + 1;
            if(i == j) j++;
            k = 0;
        }
    }
    int ans = min(i,j);
    cout << s.substr(ans,n);
}

signed main()
{
    cin.tie(0)->ios::sync_with_stdio(0);                
    cout << setiosflags(ios::fixed) << setprecision(2); 
    int TT = 1;
    //cin >> TT; 
    while (TT--)
    {
        solve(); 
        //cout << endl;
    }
}
```

## 马拉车

原文件：`算法/字符串/马拉车.cpp`

```cpp
/*
为了解决字符串最大回文问题，最重要的是回文半径数组
时间复杂度On

把原字符串扩展加上#，方便找奇长度和偶数长度的回文

回文半径和真实长度对比是a[i]
真实回文串结尾位置 = 扩展串的结尾下标 / 2

r 代表以当前为中心向右扩展，第一个到达不了的位置
c 此时取得最右边界的中心位置，而且是最早的位置

对向下的每个中心
首先分两种情况：
    一、在r之外
    对于这种直接遍历找，暴力
    二、在r之内
    这种又分三种情况
    （1）关于c对称点的半径依旧在r内
        这样直接就相等就可以
    （2）对称点半径在大回文区域以外
        a[i] = r - i
    （3）对称点半径正好撞线大回文边界
        继续在r的基础上走，暴力继续找，往外扩
*/
#include <bits/stdc++.h> 
using namespace std;
#define int long long                 
#define endl '\n'                     
#define all(x) (x).begin(), (x).end() 

constexpr int MOD = 1e9 + 7;

vector<int> a;

void get(string s)
{
    string t = "#";
    for(char k : s)
    {
        t += k;
        t += '#';
    }
    int n = t.size();
    a.assign(n+1,0);
    int r = 0;
    int c = 0;
    for(int i = 0;i < n;i++)
    {
        if(i < r)
        {
            a[i] = min(a[2 * c - i],r - i);
        }
        while(i - a[i] - 1 >= 0 && i + a[i] +1 < n && t[i-a[i]-1] == t[i+a[i]+1])
        a[i]++;
        if(i + a[i] > r)
        {
            c = i;
            r = i + a[i];
        }
    }
}

void solve()
{
    string s,ans;
    cin >> s;
    get(s);
    int c = 0;
    for(int i = 0;i < a.size();i++)
    {
        if(a[i] > a[c])
        {
            c = i;
        }
    }
    int r = a[c];
    ans = s.substr((c - r)/2,r);
    cout << ans;
}

signed main()
{
    cin.tie(0)->ios::sync_with_stdio(0);                
    cout << setiosflags(ios::fixed) << setprecision(2); 
    int TT = 1;
    //cin >> TT; 
    while (TT--)
    {
        solve(); 
        //cout << endl;
    }
}
```
