## 使用说明

这是原有学习代码的整理版。先查看下方代码中的输入输出和变量约定；题目专用实现不自动等同于通用模板。

这是最大流EK算法
对于一个图，有一个源点和一个汇点
存在一些边，这些边存在最大容量
求从源点出发，到汇点的最大流量，边上经过的流量不能超过限制
具体流程：
先bfs找增广路，也就是合理路径，mf数组记录流量上限
对于找到的增光路，将其反向建边，也就是建立残留网
将可行流累加，总体就是最大流量
复杂度 On*m*m

## 代码

```cpp
/*
这是最大流EK算法
对于一个图，有一个源点和一个汇点
存在一些边，这些边存在最大容量
求从源点出发，到汇点的最大流量，边上经过的流量不能超过限制

具体流程：
先bfs找增广路，也就是合理路径，mf数组记录流量上限
对于找到的增光路，将其反向建边，也就是建立残留网
将可行流累加，总体就是最大流量

复杂度 On*m*m
*/
#include <bits/stdc++.h> 
using namespace std;
#define int long long                 
#define endl '\n'                     
#define all(x) (x).begin(), (x).end() 

constexpr int MOD = 1e9 + 7;

const int N = 10005; // 节点数
const int M = 200005; // 边数，记得开两倍，因为有反向边

int head[N], to[M], val[M], nxt[M], cnt = 1; // cnt初值为1，使得正反向边可以通过 ^1 转换
int mf[N], pre[M]; // mf[i]表示到达点i的最大流量，pre[i]记录到达点i的边的编号
bool vis[N]; // BFS时记录点是否访问过
int n, m, s, t; // n个点，m条边，源点s，汇点t

void add(int u, int v, int w) {
    to[++cnt] = v;
    val[cnt] = w;
    nxt[cnt] = head[u];
    head[u] = cnt;
}

// 添边：正向建边容量为w，反向建边容量为0
void add_edge(int u, int v, int w) {
    add(u, v, w);
    add(v, u, 0);
}

bool bfs() {
    memset(mf, 0, sizeof(mf));
    memset(vis,false,sizeof(vis));
    queue<int> q;
    q.push(s);
    vis[s] = true;
    mf[s] = 1e18; // 初始源点流量无穷大
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        for (int i = head[u]; i; i = nxt[i]) 
        {
            int v = to[i], w = val[i];
            if (!vis[v] && w > 0) { // 如果未访问过且残余容量大于0
                vis[v] = true;
                mf[v] = min(mf[u], w);
                pre[v] = i; // 记录过来的边
                q.push(v);
                if (v == t) return true; // 找到一条增广路
            }
        }
    }
    return false;
}

int EK() {
    int maxflow = 0;
    while (bfs()) {
        int v = t;
        // 顺着路径寻找，更新容量残余网
        while (v != s) {
            int i = pre[v]; // 走到这里的边
            val[i] -= mf[t];
            val[i ^ 1] += mf[t]; // 反向边增加
            v = to[i ^ 1]; // 回溯到上一个点
        }
        maxflow += mf[t]; // 累加可行流
    }
    return maxflow;
}

void solve()
{
    // 初始化
    memset(head, 0, sizeof(head));
    cnt = 1;

    cin >> n >> m >> s >> t;
    for (int i = 1; i <= m; i++) {
        int u, v, w;
        cin >> u >> v >> w;
        add_edge(u, v, w);
    }
    
    cout << EK();
}

signed main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);                
    cout << setiosflags(ios::fixed) << setprecision(2); 
    int TT = 1;
    cin >> TT; 
    while (TT--)
    {
        solve(); 
        cout << endl;
    }
}
```
