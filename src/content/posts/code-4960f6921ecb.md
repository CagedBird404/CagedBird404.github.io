## 01背包问题

原文件：`算法/dp/01背包问题.cpp`

```cpp
/*
01背包：每个物品 要和不要 两种可能性展开

给定一个正数t，表示背包容量
有m个货品，每个可以选一次
每个货品有自己的体积和价值
返回不超过容量的情况下，怎么获得最大的价值

定义dp[i][j] ：前i个物品总花费不超过j，最大价值
*/
#include <bits/stdc++.h> 
using namespace std;
#define int long long                 
#define endl '\n'                     
#define all(x) (x).begin(), (x).end() 

constexpr int MOD = 1e9 + 7;

void solve()
{
    int t,n;
    cin >> t >> n;
    vector<vector<int>> dp(n+1,vector<int>(t+1));
    vector<int> cost(n+1),w(n+1);
    for(int i = 1;i <= n;i++) cin >> cost[i] >> w[i];
    for(int i = 1;i <= n;i++)
    {
        for(int j = 0;j <= t;j++)
        {
            dp[i][j] = dp[i-1][j];
            if(j - cost[i] >= 0)
            {
                dp[i][j] = max(dp[i][j],dp[i-1][j-cost[i]] + w[i]);
            }
        }
    }
    cout << dp[n][t];


    //压缩
    vector<int> dp1(t+1);
    for(int i = 1;i <= n;i++)
    {
        for(int j = t;j >= cost[i];j--)
        {
            dp1[j] = max(dp1[j],dp[j - cost[i]] + w[i]);
        }
    }
    cout << dp1[t];
}

signed main()
{
    cin.tie(0)->ios::sync_with_stdio(0);                
    cout << setiosflags(ios::fixed) << setprecision(2); 
    int TT = 1;
    //cin >> TT;
```

```cpp
/*
分组背包问题：多个背包分组，每组只能拿一个
             原理是每组都进行展开，注意复杂度不会变，都是O(n*t)

每个物品有三个属性：消耗，价值，组号
同一组只能选一件

和01区别就是改为要不要这组的商品，这组都依次变量
*/
#include <bits/stdc++.h> 
using namespace std;
#define int long long                 
#define endl '\n'                     
#define all(x) (x).begin(), (x).end() 

constexpr int MOD = 1e9 + 7;

struct node
{
    int a,b,c;
};

void solve()
{
    int n,t;
    cin >> t >> n;
    vector<node> f(n+1);
    for(int i = 1;i <= n;i++) cin >> f[i].a >> f[i].b >> f[i].c;
    sort(f.begin()+1,f.end(),[](node f1,node f2){
        return f1.c < f2.c;
    });
    int team = f[n].c;
    vector<vector<int>> dp(team+1,vector<int>(t+1));
    int start = 1,end = 2;
    for(int i = 1;i <= team;i++)
    {
        while(end <= n && f[start].c == f[end].c) end++;
        for(int j = 0;j <= t;j++)
        {
            dp[i][j] = dp[i-1][j];
            for(int k = start;k < end;k++)
            {
                if(j - f[k].a >= 0)
                {
                    dp[i][j] = max(dp[i][j],dp[i-1][j-f[k].a] + f[k].b);
                }
            }
        }
        start = end;
        end++;
    }
    cout << dp[team][t];


    //空间压缩
    vector<int> dp1(t+1);
    start = 1,end = 2;
    for(int i = 1;i <= team;i++)
    {
        while(end <= n && f[end].c == f[start].c) end++;
        for(int j = t;j >= 0;j--)
        {
            for(int k = start;k < end;k++)
            {
                if(j - f[k].a >= 0)
                {
                    dp1[j] = max(dp1[j],dp1[j-f[k].a] + f[k].b);
                }
            }
        }
    }
    cout << dp1[t];
}

signed main()
{
    cin.tie(0)->ios::sync_with_stdio(0);                
    cout << setiosflags(ios::fixed) << setprecision(2); 
    int TT = 1;
    //cin >> TT; 
    while (TT--)
    {
        solve(); 
        //cout << endl;
    }
}
```

## 最长上升公共子序列

原文件：`算法/dp/最长上升公共子序列.cpp`

```cpp
/*
用于找最长公共上升子序列

时间复杂度O(n*m)
*/

#include <bits/stdc++.h> 
using namespace std;
#define int long long                 
#define endl '\n'                     
#define all(x) (x).begin(), (x).end() 

constexpr int MOD = 1e9 + 7;

void solve()
{
    int n,m;
    cin >> n ;
    vector<int> a(n);
    for(int i = 0;i < n;i++) cin >> a[i];
    cin >> m;
    vector<int> b(m);
    for(int i = 0;i < m;i++) cin >> b[i];
    vector<int> dp(m),k(m,-1);   //dp[j]记录的是对于b数组以b[j]为结尾时
    for(int i = 0;i < n;i++)     //他的最大公共上升子区间长度
    {                            //k记录他的前驱元素，便于去回溯寻找整个序列
        int l = 0;//最大长度
        int num = -1;//前驱元素
        for(int j = 0;j < m;j++)
        {
            if(a[i] == b[j])  //关键限制，因为只能是相同的元素，所以只有相等
            {                 //的时候才对他加一
                if(dp[j] < l + 1)   //限制：只对于大于当前最佳长度的进行更新
                {
                    dp[j] = l + 1;
                    k[j] = num; //更新前置元素
                }
            }
            if(b[j] < a[i])  //对于比他小的b元素，如果dp[j]的长度比他大，那么就把最佳长度更新
            {                //并且更新前置
                if(dp[j] > l)
                {
                    l = dp[j];
                    num = j;
                }
            }
        }
    }
    int num = max_element(all(dp)) - dp.begin();
    int ans = dp[num];
    vector<int> pp;
    while(num != -1)
    {
        pp.emplace_back(b[num]);
        num = k[num];
    }
    reverse(all(pp));
    cout << ans << endl; if(ans == 0) return;
    for(auto gg : pp)
    {
        cout << gg << " ";
    }
}

signed main()
{
    cin.tie(0)->ios::sync_with_stdio(0);                
    cout << setiosflags(ios::fixed) << setprecision(2); 
    int TT = 1;
    //cin >> TT; 
    while (TT--)
    {
        solve(); 
        //cout << endl;
    }
}
```

```cpp
/*
求一个数列最长单调上升的子序列，不需要连续
下降同理，可以加入二分优化
*/
#include <bits/stdc++.h> 
using namespace std;
#define int long long                 
#define endl '\n'                     
#define all(x) (x).begin(), (x).end() 

constexpr int MOD = 1e9 + 7;

void solve()
{
    int n;
    cin >> n;
    vector<int> a(n);
    for(int i = 0;i < n;i++) cin >> a[i];
    vector<int> b(n+1);//上升,记录他的最小结尾,最后数组就记录了这个数组
    vector<int> dp1(n,1);
    b[0] = -1e9;
    int num = 0; //最后上升序列长度
    for(int i = 0;i < n;i++)
    {
        if(a[i] > b[num])
        {
            num++;
            b[num] = a[i];
            dp1[i] = num;
        }
        else{
            int l = 1,r = num;
            while(l <= r)
            {
                int mid = l + (r - l) / 2;
                if(b[mid] > a[i])
                {
                    r = mid - 1;
                }
                else{
                    l = mid + 1;
                }
            }
            b[l] = a[i];
            dp1[i] = l + 1;
        }
    }
    cout << num << endl;


    vector<int> c(n+1);
    vector<int> dp2(n,1);
    c[0] = 1e9;
    num = 0;
    for(int i = 0;i < n;i++)
    {
        if(a[i] < c[num])
        {
            num++;
            c[num] = a[i];
            dp2[i] = num;
        }
        else{
            int l = 1,r = num;
            while(l <= r)
            {
                int mid = l + (r - l) / 2;
                if(c[mid] < a[i])
                {
                    r = mid - 1;
                }
                else{
                    l = mid + 1;
                }
            }
            c[l] = a[i];
            dp2[i] = l + 1;
        }
    }
    cout << num;
}
```

## 树型dp

原文件：`算法/dp/树型dp.cpp`

```cpp
/*
套路：
1.分析父树得到答案需要子树的哪些信息
2.把子树信息的全集定义成递归返回值
3.通过递归让子树返回全集信息
4。整合子树的全集信息得到父树的全集信息并返回
*/
#include <bits/stdc++.h> 
using namespace std;
#define int long long                 
#define endl '\n'                     
#define all(x) (x).begin(), (x).end() 

constexpr int MOD = 1e9 + 7;
constexpr int N = 2e5 + 10;
vector<int> adj[N];
int dp[N][2]; // dp[u][0] 不选u, dp[u][1] 选u
int a[N];     // 节点权值

void dfs(int u, int p)
{
    dp[u][0] = 0;
    dp[u][1] = a[u];
    for (auto v : adj[u])
    {
        if (v == p) continue;
        dfs(v, u);
        // 状态转移方程（以最大权独立集为例：“没有上司的舞会”）
        dp[u][0] += max(dp[v][0], dp[v][1]);
        dp[u][1] += dp[v][0];
    }
}

void solve()
{
    int n;
    cin >> n;
    
    // 初始化清空
    for(int i = 1; i <= n; i++) 
    {
        adj[i].clear();
        dp[i][0] = dp[i][1] = 0;
    }
    
    // 读入点权
    for(int i = 1; i <= n; i++) cin >> a[i];
    
    // 读入边
    for(int i = 1; i < n; i++) 
    {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    
    dfs(1, 0); // 从根节点(默认1)开始DFS
    
    cout << max(dp[1][0], dp[1][1]) << endl;
}

signed main()
{
    cin.tie(0)->ios::sync_with_stdio(0);                
    cout << setiosflags(ios::fixed) << setprecision(2); 
    int TT = 1;
    //cin >> TT; 
    while (TT--)
    {
        solve(); 
        cout << endl;
    }
}
```
