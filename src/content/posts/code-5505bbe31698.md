## STL  set（去重）

原文件：`STL/STL  set（去重）.cpp`

```cpp
#include<bits/stdc++.h>//可以用来去重，排序
using namespace std;//set中一个元素在集合中只能出现一次，multiset中可以出现多次，两个都会从小到大排好序
int main(){
    set<int> st;
    st.insert(1);
    st.insert(2);
    st.insert(0);
    st.erase(2);//删除
    st.find(1);//寻找元素
    st.count(2);//查询有几个，set中返回值只有两个1或0
    if(st.count(1))
    {
        cout<<"yes"<<endl;
    };
    cout<<st.size()<<endl;
    //st.clear();//清空
    cout<<st.empty()<<endl;
    for(auto&&i:st)
    {
        cout<<i<<endl;
    };
    //迭代器进行遍历
    for(set<int>::iterator it=st.begin();it!=st.end();++it)
    {
        cout<<*it<<endl;
    };
}
```

## STL list

原文件：`STL/STL list.cpp`

```cpp
/*
list 不支持随机访问，
可以在已知位置插入删除，复杂度O1
缺点是占空间很大

*/
#include<bits/stdc++.h>
using namespace std;
int main(){
    list<int> lst = {1,2,3};
    lst.push_back(4);//结尾放
    lst.push_front(0);//开头放
    auto it = lst.begin();
    advance(it,2);//将某个迭代器移动，后面数字代表n，n>0代表向前移动，反之
                  //他是通用的，很多容器都可以用
    lst.insert(it,99);//将99送到第2个位置
    lst.erase(it);//删除当前位置
    lst.sort();//list自带函数，因为list不能用正常的sort，这个内置函数也可以用表达式
    lst.sort([](int a,int b){
        return a > b;
    });
    lst.unique();//自带的去重
    list<int> a = {1,2,3}
    auto iit = lst.begin();
    advance(it,1);
    lst.splice(it,a);//将a的元素都搬到lst的位置上
}
```

## STL search

原文件：`STL/STL search.cpp`

```cpp
/*
对于search可以在一个大容器中返回小容器出现的位置
和find类似但是这个是对于两个容器
复杂度为暴力
*/
#include <iostream>
#include <vector>
#include <algorithm>

int main() {
    std::vector<int> v = {1, 2, 3, 4, 5, 3, 4};
    std::vector<int> sub = {3, 4};

    auto it = std::search(v.begin(), v.end(), sub.begin(), sub.end());

    if (it != v.end()) {
        std::cout << "Found at position: " << it - v.begin() << std::endl;
    } else {
        std::cout << "Not found" << std::endl;
    }
}
```
