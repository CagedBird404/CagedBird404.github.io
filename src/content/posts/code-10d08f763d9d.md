/*
继承好处：减少重复代码

语法 ： class 子类: 继承方式 父类

#include<bits/stdc++.h>
using namespace std;

class gg
{
public:
    void sh()
    {
        cout << "kaitou" << endl;
    }

    {
        cout << "mulu" << endl;
    }
    void ji()
    {
        cout << "jiewei" << endl;
    }

};

class Gcc : public gg
{
public:
    void zj()
    {
        cout << "gcc shipin" << endl;
    }
};
class java : public gg
{
public:

    void zj()
    {
        cout << "java shipin" << endl;
    }

};

class python : public gg
{
public:

    void zj()
    {
        cout << "python shipin" << endl;
    }
};

void text01()
{
    java ja;
    ja.sh(),ja.mu(),ja.zj(),ja.ji();
    python py;
    py.sh(),py.mu(),py.zj(),py.ji();
    Gcc gk;
    gk.sh(),gk.mu(),gk.zj(),gk.ji();
}

int main(){
    text01();
}

/*
对于析构和构造函数
顺序是： 父类构造 -> 子类构造 -> 子类析构 -> 父类析构
*/

#include<bits/stdc++.h>
using namespace std;

//继承中的对象模型
class base
{
public:

    int a;

protected:

    int b;

private:

    int c;
};
//对象中的私有也会继承到子类中，占内存
//父类中所有非静态成员属性都会被继承到子类

class son:public base
{
public:

    int d;
};

{

    cout << sizeof(k) << endl;

## 4继承中同名的成员处理方式

原文件：`基础/面向对象/类和对象/6继承/4继承中同名的成员处理方式.cpp`

```cpp
/*
对于子类和父类同名的成员
    子类成员直接访问就可以
    父类成员要加作用域
*/

/*
对于静态成员也是同理
*/
#include<bits/stdc++.h>
using namespace std;

class base
{
public:

    int a;

    void te()
    {
        cout << 1 << endl;
    }
//子类会隐藏所有父类同名的成员函数，即使你用了重载，都必须加作用域
    void te(int )
    {
        cout << 3;
    }
    base()
    {
        a = 100;
    }

};

class son : public base
{
public:

    int a;

    void te()
    {
        cout << 2 << endl;
    }
    son()
    {
        a = 200;
    }
};

void text01()
{
    son s;
    cout << s.a << endl;
    cout << s.base::a << endl;
    s.te();
    s.base::te();
}

int main(){
    text01();

}
```

## 6菱形继承

原文件：`基础/面向对象/类和对象/6继承/6菱形继承.cpp`

```cpp
/*
菱形继承，又叫钻石继承
*/
#include<bits/stdc++.h>
using namespace std;

class animal
{
public:

    int a = 10;

};
//利用虚继承 解决菱形继承问题
//加上virtual 后animal被称为虚基类
//底层是将继承两个数据改为两个指针，sheep和tuo对yt传下去的是两个指针，他们对应一个数据
class sheep : virtual public animal
{

};

class tuo : virtual public animal
{};

class yt : public sheep , public tuo
{};

void text01()
{
    yt k;
    //这份数据 只需要一份就可以 但是菱形数据有两份 浪费资源
    
    k.sheep::a = 18;
    k.tuo::a = 28;
    cout << k.sheep::a << endl << k.tuo :: a << endl << k.a << endl;
}

int main(){
    text01();
}
```
