## 1线段树基础加法

原文件：`算法/线段树/1线段树基础加法.cpp`

```cpp
/*
线段树维护信息类型：
父范围上某个信息，可以用O1的时间，从子范围的信息加工得到
满足的：累加和，最大值，最小值
不满足：某范围出现次数最大的数

线段树经典功能（但系调用时间都为log n）
1、范围查询
2、范围修改

*/
#include <bits/stdc++.h> 
using namespace std;
#define int long long                 
#define endl '\n'                     
#define all(x) (x).begin(), (x).end() 

constexpr int MOD = 1e9 + 7;

int n, m;
vector<int> a, d;//a记录初始数组，d是分出原来的组的加和
vector<int> b; //懒标记数组

//建树
void build(int l ,int r, int p) //范围l，r，标号p
{
    if(l == r)
    {
        d[p] = a[l];
        return;
    }
    int mid = l + (r - l) / 2;
    build(l, mid, p * 2);
    build(mid + 1, r, 2 * p + 1);
    d[p] = d[p * 2] + d[p * 2 + 1];
}

void push_down(int p, int l, int r) {
    if (b[p] == 0) return;

    int mid = (l + r) >> 1;

    d[p * 2]     += b[p] * (mid - l + 1);
    d[p * 2 + 1] += b[p] * (r - mid);

    b[p * 2]     += b[p];
    b[p * 2 + 1] += b[p];

    b[p] = 0;
}


//区间修改
void update(int l, int r, int c, int s, int t, int p) //c代表加的值，s和t是当前区间
{
    if(r < s || l > t) return;
    if(l <= s && t <= r)
    {
        d[p] += c * (t - s + 1);
        b[p] += c;
        return;
    }
    int mid = s + (t - s) / 2;
    
    push_down(p, s, t);

    if(l <= mid) update(l, r, c, s, mid, p * 2);
    if(r > mid) update(l, r, c, mid + 1, t, p * 2 + 1);
    d[p] = d[p * 2] + d[p * 2 + 1];
}

//求加和
int sum(int l, int r, int s, int t, int p)
{
    if(l > t || r < s) return 0;
    if(l <= s && t <= r)
    {
        return d[p];
    }
    int mid = s + (t - s) / 2;
    push_down(p, s, t);
    int ans = 0;
    if(l <= mid) ans += sum(l, r, s, mid, p * 2);
    if(r > mid) ans += sum(l, r, mid + 1, t, p * 2 + 1);
    return ans;
}

void solve()
{
    cin >> n >> m;
    a.assign(n + 1 ,0);
    d.assign(4 * n + 1, 0);
    b.assign(4 * n + 1, 0);

    for(int i = 1;i <= n;i++) cin >> a[i];
    build(1, n, 1);

    while(m--)
    {
        int op;
        cin >> op;
        if(op == 1)
        {
            int l, r;
            cin >> l >> r;
            cout << sum(l, r, 1, n, 1) << endl;
        }
        else{
            int l, r, x;
            cin >> l >> r >> x;
            update(l, r, x, 1, n, 1);
        }
    }
}

signed main()
{
    cin.tie(0)->ios::sync_with_stdio(0);                
    cout << setiosflags(ios::fixed) << setprecision(2); 
    int TT = 1;
    cin >> TT; 
    while (TT--)
    {
        solve(); 
        cout << endl;
    }
}
```

## 2线段树加法和乘法

原文件：`算法/线段树/2线段树加法和乘法.cpp`

```cpp
#include <bits/stdc++.h> 
using namespace std;
#define int long long                 
#define endl '\n'                     
#define all(x) (x).begin(), (x).end() 

constexpr int MOD = 1e9 + 7;

int n, m;
vector<int> a, d;
vector<int> e, f;

void build(int l, int r, int p)
{
    if(l == r)
    {
        d[p] = a[l];
        f[p] = 1; //乘法懒标记初始化
        return;
    }
    int mid = l + (r - l) / 2;
    build(l, mid, p * 2);
    build(mid + 1, r, p * 2 + 1);

    d[p] = d[p * 2] + d[p * 2 + 1];
}

void push_down(int p, int l, int r)
{
    int mid = l + (r - l) / 2;

    d[2 * p] = (d[2 * p] * f[p] % MOD + e[p] * (mid - l + 1)) % MOD;
    d[2 * p + 1] = (d[2 * p + 1] * f[p] % MOD + e[p] * (r - mid)) % MOD;

    f[p * 2] = f[p * 2] * f[p] % MOD;
    f[p * 2 + 1] = f[p * 2 + 1] * f[p] % MOD;

    e[p * 2] = (e[p] + (e[p * 2] * f[p])) % MOD;
    e[p * 2 + 1] = (e[p] + (e[p * 2 + 1] * f[p])) % MOD;

    e[p] = 0;
    f[p] = 1;
}

void update1(int l, int r, int c, int s, int t, int p)
{
    if(r < s || l > t) return;

    if(l <= s && r >= t)
    {
        d[p] = (d[p] + c * (t - s + 1)) % MOD;
        e[p] = (e[p] + c) % MOD;
        return;
    }

    push_down(p, s, t);

    int mid = s + (t - s) / 2;
    if(mid >= l) update1(l , r, c, s, mid, p * 2);
    if(mid < r) update1(l, r, c, mid + 1, t, p * 2 + 1);

    d[p] = (d[p * 2 + 1] + d[p * 2]) % MOD;

}

void update2(int l, int r, int c, int s, int t, int p)
{
    if(r < s || l > t) return;

    if(l <= s && r >= t)
    {
        d[p] = (d[p] * c) % MOD;
        e[p] = e[p] * c % MOD;
        f[p] = (f[p] * c) % MOD;

        return;
    }

    push_down(p, s, t);

    int mid = s + (t - s) / 2;
    if(mid >= l) update2(l , r, c, s, mid, p * 2);
    if(mid < r) update2(l, r, c, mid + 1, t, p * 2 + 1);

    d[p] = (d[p * 2 + 1] + d[p * 2]) % MOD;

}

int sum(int l, int r, int s, int t, int p)
{
    if(r < s || l > t) return 0;

    if(l <= s && t <= r)
    {
        return d[p] % MOD;
    }

    push_down(p, s, t);

    int ans = 0;
    int mid = s + (t - s) / 2;

    if(mid >= l) ans += sum(l, r, s, mid, p * 2);
    ans %= MOD;
    if(mid < r) ans += sum(l, r, mid + 1, t, p * 2 + 1);
    ans %= MOD;

    return ans;
}

void solve()
{
    cin >> n >> m;
    a.assign(n + 1, 0);
    d.assign(4 * n + 1, 0);
    e.assign(4 * n + 1, 0);
    f.assign(4 * n + 1, 1);
    for(int i = 1;i <= n;i++) cin >> a[i];

    build(1, n, 1);

    while(m--)
    {
        int op;
        cin >> op;
        if(op == 1)
        {
            int l, r, k;
            cin >> l >> r >> k;
            update2(l, r, k, 1, n, 1);
        }
        else if(op == 2)
        {
            int l, r, k;
            cin >> l >> r >> k;
            update1(l, r, k, 1, n, 1);
        }
        else{
            int l, r;
            cin >> l >> r;
            cout << sum(l, r, 1, n, 1) << endl;
        }
    }
}

signed main()
{
    cin.tie(0)->ios::sync_with_stdio(0);                
    cout << setiosflags(ios::fixed) << setprecision(2); 
    int TT = 1;
    cin >> TT; 
    while (TT--)
    {
        solve(); 
        cout << endl;
    }
}
```

## 线段树(废物版)

原文件：`算法/线段树/线段树(废物版).cpp`

```cpp
/*
这是我刚学算法时，学到递归，
突发奇想写的废物线段树
并无用处，留作纪念
*/

/*
线段树
首先需要建树
需要想好每个节点所需要的信息
和分治一样，先想好分的过程，再想如何合
建立树的过程为 自低向上分治计算合并    时间复杂度O(n)
查询的过程为   自项先向下分治查找合并  时间复杂度O(log(n))
*/
#include <bits/stdc++.h>
using namespace std;
#define int long long
#define endl '\n'
#define all(x) (x).begin(), (x).end()

constexpr int MOD = 1e9 + 7;

struct Node // 先建立节点，节点中的元素都和规则有关
{
    int sum;  // 区间总和
    int qmax; // 最大前缀和
    int hmax; // 最大后缀和
    int Max;  // 最大子区间和
};

vector<int> a;    // 原数组
vector<Node> seg; // 线段树

Node merge(Node L, Node R) // 合并两个分支,merge一般代表合并区间的意思
{                          // 要找出合并原则
    if (L.sum == LLONG_MIN / 2)
        return R;
    if (R.sum == LLONG_MIN / 2)
        return L;
    Node b;
    b.sum = L.sum + R.sum;
    b.qmax = max(L.qmax, L.sum + R.qmax);
    b.hmax = max(R.hmax, L.hmax + R.sum);
    b.Max = max({L.Max, R.Max, L.hmax + R.qmax});
    return b;
}

Node build(int idx, int l, int r) // 建立树，idx代表序号
{                                 // 建树就是把分支已经细化成单个元素了，再返回节点里的元素
    if (l == r)
    {
        int val = a[l];
        return seg[idx] = {val, val, val, val};
    }
    int mid = l + (r - l) / 2;
    Node L = build(idx * 2, l, mid);         // 代表向下分支的左边
    Node R = build(idx * 2 + 1, mid + 1, r); // 代表向下分支的右边
    return seg[idx] = merge(L, R);
}

void update(int idx, int l, int r, int pos, int val) // 将第pas个数修改成val，再恢复树
{
    if (l == r)
    {
        seg[idx] = {val, val, val, val};
        return;
    }
    int mid = l + (r - l) / 2;
    if (pos > mid)
        update(idx * 2 + 1, mid + 1, r, pos, val);
    else
        update(idx * 2, l, mid, pos, val);
    seg[idx] = merge(seg[idx * 2], seg[idx * 2 + 1]); // 修复
}

Node tree(int idx, int l, int r, int ql, int qr)
{
    if (l > qr || r < ql) // 不在范围内
    {
        return {LLONG_MIN / 2, LLONG_MIN / 2, LLONG_MIN / 2, LLONG_MIN / 2};
    }
    if (l >= ql && r <= qr)
    {
        return seg[idx];
    }
    int mid = l + (r - l) / 2;
    Node L = tree(idx * 2, l, mid, ql, qr);
    Node R = tree(idx * 2 + 1, mid + 1, r, ql, qr);
    return merge(L, R);
}
void solve()
{

    int n;
    cin >> n;
    a.resize(n + 1);
    seg.resize(4 * n + 5);
    for (int i = 1; i <= n; i++)
        cin >> a[i];
    build(1, 1, n);
    int q;
    cin >> q;
    while (q--)
    {
        int l, r;
        cin >> l >> r;
        Node ans = tree(1, 1, n, l, r);
        cout << ans.Max << endl;
    }
}
// 将每组测试用例的处理逻辑写在里面

signed main()
{
    cin.tie(0)->ios::sync_with_stdio(0);
    cout << setiosflags(ios::fixed) << setprecision(2);
    int TT = 1;
    // cin >> TT;
    while (TT--)
    {
        solve();
        cout << endl;
    }
}
```
