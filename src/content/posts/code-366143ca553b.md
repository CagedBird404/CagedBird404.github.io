## 封装1

原文件：`基础/面向对象/类和对象/1封装/封装1.cpp`

```cpp
/*
c++面向对象三大特征：封装，继承，多态
c++认为 万事万物都皆为对象，对象上有其属性和行为
具有相同性质的对象，我们抽象为类

封装的意义：
* 将属性和行为作为一个整体，表现生活中的事物
* 将属性和行为加以权限控制

一、
    在设计类的时候，属性和行为写在一起，表现事物
语法： class 类名{ 访问权限： 属性 / 行为};
*/
//实际化：通过一个类 创建一个对象
#include<bits/stdc++.h>
using namespace std;
const double PI = 3.14;

//例1：设计一个圆类，求圆周长
class yuan{
    //访问权限：
public:

    //属性
    //半径
    int r;

    //行为
    //获取圆的周长
    double zhou()
    {
        return 2 * PI * r;
    }
};

//例2:设计学生类
class student{
public:
//类中的属性和行为我们统一称为 成员
//属性 也叫 成员属性 或者 成员变量
//行为 也叫 成员函数 或者 成员方法
    string m_name;
    int id;
    void show()
    {
        cout << m_name << " " << id << endl;
    }
    void setname(string s)
    {
        m_name = s;
    }
    void setid(int a)
    {
        id = a;
    }
};
int main()
{
    //通过圆类，创建具体的圆（对象）
    yuan c1;
    c1.r = 4;
    cout << c1.zhou() << endl;
    student m1;
    m1.m_name = "张三";
    m1.id = 2504;
    m1.show();
    student m2;
    m2.setname("李四");
    m2.setid(999);
    m2.show();
}
```

```cpp
/*
三种权限：
1、public    公共权限    成员 类内可以访问  类外可以访问
2、protected 保护权限    成员 类内可以访问  类外不可以访问    儿子可以访问父亲中保护的内容
3、private   私有权限    成员 类内可以访问  类外不可以访问    儿子不可以访问父亲的私有内容

struct 和 class区别：
struct 默认是 公共 public
class  默认是 私有 private
*/
#include<bits/stdc++.h>
using namespace std;

class Person
{
public:
    string name;

protected:
    string car_name;

private:
    int passworld;

public:
    void func()
    {
        name = "张三";
        car_name = "拖拉机";
        passworld = 20000;
    }
};

int main()
{
    Person p1;
    p1.name = "李四";
    // p1.car_name = "奔驰";   类外不可以更改
}
```

```cpp
/*
成员属性设置私有
1、可以自己控制读写权限
2、对于写可以检测数据有效性
*/
#include<bits/stdc++.h>
using namespace std;

class Person
{
private:
    string name;

    int Age = 18;

    string idol;

public:
    void setname(string s)//可读可写
    {
        name = s;
    }
    string getname()
    {
        return name;
    }
    int getage()   //只读状态，不能输入
    {              //只写状态，不能输出
        return Age;
    }
    int setAge(int a)
    {
        if(a > 0 && a <= 150) Age = a;
        else cout << "年龄" << a << "输入有误，赋值失败"<<endl;
        return 0;
    }
};
int main()
{
    Person p;
    p.setname("张三");
    cout << p.getname() << " " << p.getage();
    int a;
    cin >> a;
    p.setAge(a);
}
```
