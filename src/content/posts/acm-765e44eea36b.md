## 代码

```cpp
#include<bits/stdc++.h>
using namespace std;
int mainn(){
    int x,a,b;
    //根号
    sqrt(x);
    //绝对值
    abs(x);
    fabs(x);//输出浮点数
    //向上取整
    ceil(x);
    //向下取整
    floor(x);
    //四舍五入
    round(x);
    //幂函数
    pow(a,b);
}
```
