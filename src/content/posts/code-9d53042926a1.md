## 1树状数组（范围查询，单点修改）

原文件：`算法/数状数组/1树状数组（范围查询，单点修改）.cpp`

```cpp
/*
在一维数组上实现单点或范围增加，单点或范围查询

二维数组上实现单点或者范围增加，范围查询

树状数组都是来维护可差分信息，线段树是用来对于不可差分信息的
什么是可差分信息：区域和，乘积
不可：最大值最小值

线段树大多数都是可以代替树状数组的
线段树优势：用法全面，思考量低，维护信息多
线段树劣势：代码量大，使用空间大，常熟时间较大
树状数组优势：代码量少，使用空间少，常数时间优异

树状数组下标必须从1开始
怎么确定树状数组下标所对应的范围，将下标二进制最右1去掉，再加1，到当前下标就是所包围范围

对于每次查询，对于l和r的二进制，通过不断去掉最右侧的1，加上对应下标的树状数组和，来进行求解
对于单点修改，通过不断加上二进制最右侧的1，来得到有哪些树状数组下标拥有这个数
对于最右侧的1，用i & -i
和线段树一样都是nlogn
*/
#include <bits/stdc++.h> 
using namespace std;
#define int long long                 
#define endl '\n'                     
#define all(x) (x).begin(), (x).end() 

constexpr int MOD = 1e9 + 7;

void solve()
{
    int n, m;
    cin >> n >> m ;
    vector<int> a(n+1), tree(n + 1);
    for(int i = 1; i <= n; i++)
    {
        cin >> a[i];
        int k = i;
        while(k <= n)
        {
            tree[k] += a[i];
            k += k & -k;
        }
    }
    while(m--)
    {
        int op, x, y;
        cin >> op >> x >> y;
        if(op == 1)
        {
            while(x <= n)
            {
                tree[x] += y;
                x += x & -x;
            }
        }
        else{
            int ans = 0;
            x--;
            while(y > 0)
            {
                ans += tree[y];
                y -= y & -y;
            }
            while(x > 0)
            {
                ans -= tree[x];
                x -= x & -x;
            }
            cout << ans << endl;
        }
    }
}

signed main()
{
    cin.tie(0)->ios::sync_with_stdio(0);                
    cout << setiosflags(ios::fixed) << setprecision(2); 
    int TT = 1;
    //cin >> TT; 
    while (TT--)
    {
        solve(); 
        //cout << endl;
    }
}
```

## 2树状数组（单点查询，范围修改）

```cpp
/*
原理就是将数组改为差分数组
利用差分数组单点修改就能改范围值的特性
*/
#include <bits/stdc++.h> 
using namespace std;
#define int long long                 
#define endl '\n'                     
#define all(x) (x).begin(), (x).end() 

constexpr int MOD = 1e9 + 7;

void solve()
{
    int n, m;
    cin >> n >> m;
    vector<int> a(n + 1), b(n + 1), tree(n + 1);
    for(int i = 1; i <= n;i++)
    {
        cin >> a[i];
        b[i] = a[i] - a[i - 1];
        int k = i;
        while(k <= n)
        {
            tree[k] += b[i];
            k += k & -k;
        }
    }
    while(m--)
    {
        int op;
        cin >> op;
        if(op == 1)
        {
            int x, y, k;
            cin >> x >> y >> k;
            while(x <= n)
            {
                tree[x] += k;
                x += x & -x;
            }
            y++;
            while(y <= n)
            {
                tree[y] -= k;
                y += y & -y;
            }
        }
        else{
            int t;
            cin >> t;
            int ans = 0;
            while(t > 0)
            {
                ans += tree[t];
                t -= t & -t;
            }
            cout << ans << endl;
        }
    }
}

signed main()
{
    cin.tie(0)->ios::sync_with_stdio(0);                
    cout << setiosflags(ios::fixed) << setprecision(2);
```

## 3树状数组（范围查询，范围修改）

原文件：`算法/数状数组/3树状数组（范围查询，范围修改）.cpp`

```cpp
/*
这种情况其实线段树最优，但是树状数组不是不能写
只是不好理解

维护两个差分，一个di,一个（i- 1） * di
求l到r的和，和单点求值一样，求1到r的减去1到l-1的
算的就是 l * di - (i - 1) * di;
*/
#include <bits/stdc++.h> 
using namespace std;
#define int long long                 
#define endl '\n'                     
#define all(x) (x).begin(), (x).end() 

constexpr int MOD = 1e9 + 7;

void solve()
{
    int n, m;
    cin >> n >> m;
    vector<int> a(n + 1), b(n + 1);
    for(int i = 1; i <= n; i++)
    {
        cin >> a[i];
        b[i] = a[i] - a[i - 1];
    }
    vector<int> tree1(n + 1), tree2(n + 1);
    for(int i = 1; i <= n; i++)
    {
        int k = i;
        while(k <= n)
        {
            tree1[k] += b[i];
            tree2[k] += (i - 1) * b[i];
            k += k & -k;
        }
    }
    while(m--)
    {
        int op;
        cin >> op;
        if(op == 1)
        {
            int x, y, k;
            cin >> x >> y >> k;
            int k1 = k * (x - 1);
            int k2 = k * y;
            y++;
            while(x <= n)
            {
                tree1[x] += k;
                tree2[x] += k1;
                x += x & -x;
            }
            while(y <= n)
            {
                tree1[y] -= k;
                tree2[y] -= k2;
                y += y & -y;
            }
        }
        else{
            int x, y;
            cin >> x >> y;
            int n1= x - 1, n2 = y;
            x--;
            int ans = 0;
            while(x > 0)
            {
                ans -= n1 * tree1[x] - tree2[x];
                x -= x & -x;
            }
            while(y > 0)
            {
                ans += n2 * tree1[y] - tree2[y];
                y -= y & -y;
            }
            cout << ans << endl;
        }
    }
}

signed main()
{
    cin.tie(0)->ios::sync_with_stdio(0);                
    cout << setiosflags(ios::fixed) << setprecision(2); 
    int TT = 1;
    //cin >> TT; 
    while (TT--)
    {
        solve(); 
        //cout << endl;
    }
}
```
