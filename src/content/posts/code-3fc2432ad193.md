## 并查集

原文件：`算法/并查集/并查集.cpp`

```cpp
// #pragma GCC optimize(2)
#include <bits/stdc++.h>
using namespace std;
#define int long long
#define endl '\n'
#define all(x) (x).begin(), (x).end()

constexpr int MOD = 1e9 + 7;
// -9.2e18 ~ 9.2e18
constexpr int Max = 2*1e5;
int fa[Max];

int find(int x)
{
    return x == fa[x] ? x : fa[x] = find(fa[x]);
}
void cc(int f,int v)
{
    fa[find(f)] = find(v);
}
bool dd(int x,int y)
{
    return find(x) == find(y);
}
void solve()
{
    int n,m,q;
    cin >> n >> m >> q;
    for(int i = 0;i <= n;i++)
    {
        fa[i] = i;
    }
    while(m--)
    {
        int x,y;
        cin >> x >> y;
        cc(x,y);
    }
    while(q--)
    {
        int x,y;
        cin >> x >> y;
        if(dd(x,y)) cout << "YES" << endl;
        else cout << "NO" << endl;
    }
}

signed main()
{
    cin.tie(0)->ios::sync_with_stdio(0);
    cout << setiosflags(ios::fixed) << setprecision(2);
    int TT = 1;
    cin >> TT;
    while (TT--)
    {
        solve();
        cout << endl;
    }
}
```

## 最小生成树（Kruskal + 并查集）

原文件：`算法/并查集/最小生成树（Kruskal + 并查集）.cpp`

```cpp
#include <bits/stdc++.h> 
using namespace std;
#define int long long                 
#define endl '\n'                     
#define all(x) (x).begin(), (x).end() 

constexpr int MOD = 1e9 + 7;
constexpr int Max = 1e5+5;

struct node
{
    int v,f,w;
};

vector<node> b(Max);
int fa[Max];

int find(int x)
{
    return x == fa[x] ? x : fa[x] = find(fa[x]);
}
void solve()
{
    int n,m;
    cin >> n >> m;
    for(int i = 1;i <= n;i++) fa[i] = i;
    for(int i = 0;i < m;i++)
    {
        cin >> b[i].v >> b[i].f >> b[i].w;
    }
    sort(b.begin(),b.begin()+m,[](node a,node b){
        return a.w < b.w;
    });
    int num = 0;
    int ans = 0;
    for(int i = 0;i < m;i++)
    {
        int fx = find(b[i].v);
        int fy = find(b[i].f);
        if(fx != fy)
        {
            num++;
            ans += b[i].w;
            fa[fx] = fy;
        }
        if(num == n-1) break;
    }
    if(num == n-1)cout << ans;
    else cout << "orz";
}
```

## 权值并查集

原文件：`算法/并查集/权值并查集.cpp`

```cpp
#include <bits/stdc++.h> 
using namespace std;
#define int long long                 
#define endl '\n'                     
#define all(x) (x).begin(), (x).end() 

constexpr int MOD = 1e9 + 7;
constexpr int Max = 200005;
int fa[Max];
int v[Max];
int find(int x)
{
    if(x != fa[x])
    {
        int f = fa[x];
        fa[x] = find(fa[x]);
        v[x] += v[f];    //重点
    }
    return fa[x];   
}
void solve()
{
    int n,m;
    while(cin >> n >> m)
    {for(int i = 0;i <=n ;i++)
    {
        fa[i] = i;
        v[i] = 0;
    }
    int ans = 0;
    while(m--)
    {
        int a,b,c;
        cin >> a >> b >> c;
        a--;
        int fx = find(a);
        int fy = find(b);
        if(fx != fy)
        {
            fa[fx] = fy;
            v[fx] = v[b] + c - v[a];
        }
        else{
            if(v[a] - v[b] != c) ans++;
        }
    }
    cout << ans<<endl;
}
}

signed main()
{
    cin.tie(0)->ios::sync_with_stdio(0);                
    cout << setiosflags(ios::fixed) << setprecision(2); 
    int TT = 1;
    //cin >> TT; 
    while (TT--)
    {
        solve(); 
        //cout << endl;
    }
}
```
